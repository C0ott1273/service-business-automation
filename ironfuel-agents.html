Macro Engine Agent
// agents/macro-engine.ts export async function macroEngineAgent(userId: string) { // 1. Gate check via Data Guardian const valid = await dataGuardian.validate(userId) if (!valid) return { status: 'insufficient_data' } // 2. Fetch rolling window const logs7d = await db.bodyweightLogs .where({ userId, date: gte(daysAgo(7)) }) // 3. Compute trend const avg7 = mean(logs7d.map(l => l.weight_lb)) const avg14 = mean(prev7.map(l => l.weight_lb)) const roc = ((avg14 - avg7) / avg14) * 100 // 4. Get strength modifier const strengthDelta = await strengthSentinel.check(userId) // 5. Apply rules let delta = 0 if (roc > 1.2) delta = +100 else if (roc < 0.3) delta = -100 if (strengthDelta.declines >= 2) delta += 75 // 6. Queue for user approval — NEVER auto-apply await db.macroSuggestions.insert({ userId, delta, roc, status: 'pending_user_approval', protein_g: current.protein_g // locked }) await pushNotification.send(userId, { title: 'Weekly check-in ready', body: `Algorithm suggests ${delta > 0 ? '+' : ''}${delta} kcal` }) }
AI Coach Agent (OpenAI Proxy)
// agents/ai-coach.ts export async function aiCoachAgent(userId: string) { // Gather context — NO user data in client const ctx = await buildUserContext(userId) const prompt = ` You are a performance nutrition assistant. User data this week: - Weight trend: ${ctx.roc}%/week - Protein adherence: ${ctx.proteinPct}% - Strength retention: ${ctx.srs}% - Compound declines: ${ctx.declines} - Calorie compliance: ${ctx.caloriePct}% - RPE trend: ${ctx.rpeTrend} Rules: - Suggest adjustments conservatively - NEVER auto-apply changes - Explain reasoning briefly - Do not invent data - Max 120 words ` const res = await openai.chat.completions.create({ model: 'gpt-4o', messages: [{ role: 'user', content: prompt }], max_tokens: 200, temperature: 0.3 // conservative }) await db.aiFeed.insert({ userId, agentTag: 'ai_coach', content: res.choices[0].message.content, context_snapshot: ctx, // audit trail requiresAction: false }) }
